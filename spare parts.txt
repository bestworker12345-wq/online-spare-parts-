    <!-- JAVASCRIPT FOR DYNAMIC FILTERING -->
    <script>
        // Mock data mapping makes to models
        const brandData = {
            "Yamaha": ["YZF-R1", "MT-07", "FZ25"],
            "Honda": ["CBR1000RR", "CB300R", "Activa"],
            "Royal Enfield": ["Classic 350", "Interceptor 650", "Himalayan"]
        };

        const yearSelect = document.getElementById('bike-year');
        const makeSelect = document.getElementById('bike-make');
        const modelSelect = document.getElementById('bike-model');
        const searchBtn = document.getElementById('search-btn');

        // Step 1: When Year is selected, enable Make
        yearSelect.addEventListener('change', function() {
            if (this.value) {
                makeSelect.disabled = false;
                // Populate makes
                makeSelect.innerHTML = '<option value="">Select Make</option>';
                Object.keys(brandData).forEach(make => {
                    makeSelect.innerHTML += <option value="${make}">${make}</option>;
                });
            } else {
                resetFilters(true, true, true);
            }
        });

        // Step 2: When Make is selected, enable Model
        makeSelect.addEventListener('change', function() {
            if (this.value) {
                modelSelect.disabled = false;
                modelSelect.innerHTML = '<option value="">Select Model</option>';
                
                // Populate models based on chosen make
                const models = brandData[this.value];
                models.forEach(model => {
                    modelSelect.innerHTML += <option value="${model}">${model}</option>;
                });
            } else {
                resetFilters(false, true, true);
            }
        });

        // Step 3: When Model is selected, enable Search Button
        modelSelect.addEventListener('change', function() {
            searchBtn.disabled = !this.value;
        });

        // Search Action
        searchBtn.addEventListener('click', function() {
            alert(Searching parts for: ${yearSelect.value} ${makeSelect.value} ${modelSelect.value});
        });

        function resetFilters(resetMake, resetModel, disableBtn) {
            if (resetMake) {
                makeSelect.innerHTML = '<option value="">Select Make</option>';
                makeSelect.disabled = true;
            }
            if (resetModel) {
                modelSelect.innerHTML = '<option value="">Select Model</option>';
                modelSelect.disabled = true;
            }
            if (disableBtn) {
                searchBtn.disabled = true;
            }
        }
    </script>
